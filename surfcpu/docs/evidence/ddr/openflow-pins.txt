-- probe.v
  module probe (
      input  wire       clk100,
      output wire       a0,        // SSTL135 output, SLEW FAST
      output wire       a1,        // SSTL135 output, default slew
      inout  wire       dq0,       // SSTL135 inout, IN_TERM 40
      inout  wire       dq1,       // SSTL135 inout, no IN_TERM
      inout  wire       dqs_p, dqs_n,  // DIFF_SSTL135 inout, IN_TERM 40
      output wire       ck_p, ck_n,    // DIFF_SSTL135 output
      output wire       led
  );
      reg [3:0] c = 0;
      always @(posedge clk100) c <= c + 1;
      assign a0 = c[0];
      assign a1 = c[1];
      wire i0, i1, is;
      IOBUF b0 (.IO(dq0), .I(c[2]), .T(c[3]), .O(i0));
      IOBUF b1 (.IO(dq1), .I(c[3]), .T(c[2]), .O(i1));
      IOBUFDS bs (.IO(dqs_p), .IOB(dqs_n), .I(c[1]), .T(c[0]), .O(is));
      OBUFDS oc (.I(c[0]), .O(ck_p), .OB(ck_n));
      assign led = i0 ^ i1 ^ is;
  endmodule

Reading: a0 and every DQ/DQS/CK pin with SLEW FAST get SSTL135_SSTL15.SLEW.FAST; a1 without it
gets SLEW.SLOW; dq0 and the DQS pair with IN_TERM UNTUNED_SPLIT_40 get IN_TERM.UNTUNED_SPLIT_40,
dq1 without it gets none; the DQS pair is a differential input (IN_DIFF); bank 34's HCLK tile
gets VREF.V_675_MV, half of the DDR3L's 1.35 V, although the XDC reader ignores INTERNAL_VREF.
The DDR designs' arty_a7_ddr3_pins.xdc uses exactly these options (WIRING.md section 2.5).
