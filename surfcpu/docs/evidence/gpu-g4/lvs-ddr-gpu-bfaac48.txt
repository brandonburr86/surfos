---------------------------------------------------------------- build/g4-lvs/lvs-solver.sh
#!/bin/sh
# The open flow's LVS proof of one build, run by hand with lvs_equiv's --solver route (one solver
# process per question) instead of the flow's single incremental libz3 session, whose memory grows
# with the run. The same arguments as tools/openxc7-build.sh's lvs-prove step, on that build's own
# files (lvs/gold.v, lvs/fabric.v after bramwe, placement.json, the synthesized JSON), plus
#   --solver 'python3 -I .../z3cmd.py' --format smt2
# Run inside the openxc7 image, through the session's capped docker wrapper, from the worktree:
#   docker run --rm -v "$PWD:/work" surfcpu-openxc7:2026-10-15 \
#     sh /work/surfcpu/build/g4-lvs/lvs-solver.sh DESIGN [LIMIT_SECONDS]
# Outputs in build/g4-lvs/DESIGN/: check-lvs.txt (lvs_equiv's report, then its exit status),
# questions.log (one line a question: bytes, seconds, answer), time.txt (/usr/bin/time -v of
# lvs_equiv), mem-peak.txt (the container's memory.current at its highest, sampled every 5 s),
# cgroup-peak.txt (memory.peak), versions.txt.
set -u
design=$1
limit=${2:-0}
D=/work/surfcpu/build/fpga/arty_a7_100t/$design/openxc7
O=/work/surfcpu/build/g4-lvs/$design
mkdir -p "$O"
export Z3CMD_LOG="$O/questions.log"
: > "$Z3CMD_LOG"
{
  echo "lvs-solver.sh $design, limit $limit s, start $(date -u +%Y-%m-%dT%H:%M:%SZ)"
  python3 -I -c 'import ctypes; l = ctypes.CDLL("libz3.so.4"); a = [ctypes.c_uint() for _ in range(4)]; l.Z3_get_version(*[ctypes.byref(x) for x in a]); print("libz3 %d.%d.%d.%d" % tuple(x.value for x in a))'
  cat /opt/openxc7/share/openxc7/source-nextpnr.txt
  python3 --version
  echo "memory.max $(cat /sys/fs/cgroup/memory.max 2>/dev/null)"
} > "$O/versions.txt" 2>&1
# the front end on three questions of known answers, before the proof
t=$(mktemp -d)
printf '(set-logic QF_UF)\n(declare-const n1 Bool)\n(define-fun n2 () Bool (and n1 (not n1)))\n(assert n2)\n(check-sat)\n' > "$t/unsat.smt2"
printf '(set-logic QF_UF)\n(declare-const n1 Bool)\n(declare-const n2 Bool)\n(define-fun n3 () Bool (and n1 (not n2)))\n(assert n3)\n(check-sat)\n' > "$t/sat.smt2"
printf '(set-logic QF_UF)\n(assert n9)\n(check-sat)\n' > "$t/error.smt2"
for q in unsat sat error; do
  a=$(Z3CMD_LOG= python3 -I /work/surfcpu/build/g4-lvs/z3cmd.py "$t/$q.smt2" 2>/dev/null); s=$?
  echo "self-test $q: answer '${a}', exit $s" >> "$O/versions.txt"
done
rm -rf "$t"
( peak=0; while :; do c=$(cat /sys/fs/cgroup/memory.current 2>/dev/null || echo 0)
    if [ "$c" -gt "$peak" ]; then peak=$c; echo "$peak" > "$O/mem-peak.txt"; fi; sleep 5; done ) &
sampler=$!
/usr/bin/time -v -o "$O/time.txt" timeout "$limit" lvs_equiv --gold "$D/lvs/gold.v" \
  --gold-top surfcpu_arty_a7_top --gate "$D/lvs/fabric.v" --gate-top fabric \
  --placement "$D/placement.json" --gold-json "$D/surfcpu_arty_a7_top.json" \
  --db /opt/openxc7/share/prjxray-db/artix7 --device xc7a100t \
  --solver "python3 -I /work/surfcpu/build/g4-lvs/z3cmd.py" --format smt2 > "$O/check-lvs.txt" 2>&1
st=$?
kill "$sampler" 2>/dev/null
cat /sys/fs/cgroup/memory.peak > "$O/cgroup-peak.txt" 2>/dev/null || true
echo "lvs-solver.sh: exit $st, end $(date -u +%Y-%m-%dT%H:%M:%SZ)" >> "$O/check-lvs.txt"
exit "$st"
---------------------------------------------------------------- build/g4-lvs/z3cmd.py
#!/usr/bin/env python3
"""A solver command for lvs_equiv --solver (its one-process-per-question route), for the openxc7
image, which has the Z3 library (libz3-4, the one lvs_equiv's default --solver libz3 links) but no
solver program. lvs_equiv writes each question as an SMT-LIB 2 file (its whole network, the
question asserted, check-sat) and passes the file's path; this hands the text to libz3's own
SMT-LIB 2 front end (Z3_eval_smtlib2_string, through ctypes) and prints Z3's answer: sat, unsat or
unknown. Anything else (a parse error, an API error, no answer) prints nothing and exits 1, which
lvs_equiv counts as unknown, never as proved. With Z3CMD_LOG set, each question appends a line:
the file's bytes, the seconds this took, the answer."""
import ctypes
import os
import sys
import time

t0 = time.monotonic()
lib = ctypes.CDLL("libz3.so.4")
lib.Z3_mk_config.restype = ctypes.c_void_p
lib.Z3_del_config.argtypes = [ctypes.c_void_p]
lib.Z3_mk_context.restype = ctypes.c_void_p
lib.Z3_mk_context.argtypes = [ctypes.c_void_p]
lib.Z3_set_error_handler.argtypes = [ctypes.c_void_p, ctypes.c_void_p]
lib.Z3_get_error_code.restype = ctypes.c_int
lib.Z3_get_error_code.argtypes = [ctypes.c_void_p]
lib.Z3_eval_smtlib2_string.restype = ctypes.c_char_p
lib.Z3_eval_smtlib2_string.argtypes = [ctypes.c_void_p, ctypes.c_char_p]

with open(sys.argv[1], "rb") as f:
    text = f.read()
cfg = lib.Z3_mk_config()
ctx = lib.Z3_mk_context(cfg)
lib.Z3_del_config(cfg)
lib.Z3_set_error_handler(ctx, None)       # no handler: an error only sets the error code
out = lib.Z3_eval_smtlib2_string(ctx, text)
err = lib.Z3_get_error_code(ctx)
ans = out.decode("utf-8", "replace").strip() if out is not None else ""

log = os.environ.get("Z3CMD_LOG")
if log:
    with open(log, "a") as f:
        f.write("%d %.3f %s %s\n" % (len(text), time.monotonic() - t0, err, ans.split("\n")[0][:60] or "-"))
if err != 0 or ans not in ("sat", "unsat", "unknown"):
    sys.exit(1)
print(ans)
